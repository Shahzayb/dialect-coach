# Exact pins, not ranges: a free-tier Space that rebuilds unattended must build the
# same thing next month as it does today. Versions verified against PyPI 2026-08-17.

# UI. st.audio_input requires >= 1.41.
streamlit==1.61.1

# Pronunciation assessment. Prosody + syllable granularity require >= 1.35; >= 1.40 for
# the JSON-configured assessment path this project uses.
azure-cognitiveservices-speech==1.51.1

# Current Google GenAI SDK (`from google import genai`).
# NOT google-generativeai — that package reached end-of-life 2025-11-30.
google-genai==2.18.1

# Audio conversion to 16 kHz / 16-bit / mono PCM WAV. Needs system ffmpeg (packages.txt)
# and the stdlib audioop module, which is why Python is pinned to 3.12 (see .python-version).
pydub==0.25.1

# Local .env loading; on Hugging Face secrets arrive as environment variables instead.
python-dotenv==1.2.3

# The progress view's frames and chart specs. Both arrive transitively with streamlit, but
# `progress_view.py` imports them directly, and a direct import deserves its own pin rather
# than relying on whatever streamlit happens to resolve next month. Versions read out of the
# built image on 2026-08-19, not recalled. (`numpy` was deliberately unpinned here until
# v0.10.0, on the grounds that nothing imported it directly. `acoustics.py` now does, so it
# is pinned below under the same rule that pinned these two.)
pandas==3.0.5
altair==6.2.2

# Structured-output schemas for the coaching model.
pydantic==2.13.4

# Formant, F0 and intensity analysis: Praat, wrapped. `acoustics.py` is the only module that
# imports it. Verified against PyPI 2026-08-20 — 0.4.7 is the latest, and it publishes NO
# linux-aarch64 wheel at any Python version, so on this arm64 container pip builds it from the
# sdist. That compile is why the Dockerfile is multi-stage; see the comment at the top of it.
#
# Chosen over scipy-plus-hand-rolled-LPC for one reason that is not accuracy: parselmouth also
# reaches Praat's Manipulation machinery (PSOLA resynthesis, PitchTier replacement, DurationTier
# time-scaling), which v0.11.0's modified-self-voice surface is built on. scipy analyses and
# cannot resynthesise. NOTE for that chunk: those are NOT bound as typed Python classes — the
# bindings ship Formant/Pitch/Intensity/Sound and no Manipulation — so v0.11.0 goes through the
# untyped `parselmouth.praat.call(...)`. Checked in the 0.4.7 sdist, not assumed.
praat-parselmouth==0.4.7

# Lexical stress. Azure returns syllable spans and scores but NO stress marks, so reduction and
# stress placement are unmeasurable without a dictionary. Pure ARPABET data with 0/1/2 stress
# digits — no nltk, no model download, no compiler, unlike g2p-en. Verified against PyPI
# 2026-08-20. Coverage measured rather than assumed: 128/128 unique words of the benchmark
# passage, and 100/101 words align against the committed Azure fixtures.
cmudict==1.1.3

# Arrives transitively with pandas and with parselmouth, but `acoustics.py` imports it directly
# to build and measure sample arrays — and this file's own rule is that a direct import deserves
# its own pin rather than whatever a transitive dependency happens to resolve next month. This
# is the same reasoning pandas and altair are pinned under. Version read out of the built image
# on 2026-08-20, not recalled.
numpy==2.5.2

# Test runner. In the one manifest rather than a separate requirements-dev.txt: the image
# is a local dev container now that hosting is not a goal, and a second manifest is a
# second thing to drift. Verified against PyPI 2026-08-18.
pytest==9.1.1

# Formatter and linter, configured in pyproject.toml. Here for the same reason pytest is —
# one manifest — and so `make lint` and CI run the same binary as each other. Verified
# against PyPI 2026-08-20.
ruff==0.16.3

# Type checker, configured in pyproject.toml. Same reasoning as ruff and pytest: one
# manifest, one pinned binary, and `make typecheck` runs what CI runs. Verified against
# PyPI 2026-08-20.
mypy==2.3.1
